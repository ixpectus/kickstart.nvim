-- 1. cache: таблица-конструктор
local cache = {
  __call = function(self, ...)
    -- self = Cache (metatable экземпляра)
    local obj = setmetatable({data = {}}, self)
    return obj
  end,
}

-- 2. Cache: метатаблица = cache
local Cache = setmetatable({data = {}}, cache)

-- 3. Cache.__index = Cache (методы)
Cache.__index = Cache
function Cache:print(val)
  print(val)
end

-- 4. c = Cache() → новый объект
local c = Cache()
c:print '3'
